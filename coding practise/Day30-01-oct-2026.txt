# Kubernetes Tasks: Create Nginx Pod

kind: pod
metadata:
  name: pathnex-pod
spec:
  container:
    - name: web
      image: nginx:latest
      ports:
        - containerPort: 80

# Git Integration

# Jenkins Pipline - Checkout Git

pipline {
   agent any
   stages {
       stage('Checkout') {
           steps {
               git branch: 'main' , url: 'https://github.com/Pathnex/sample-java-app.git'
           }
       }
       stage('List Files') {
           steps {
               sh 'ls -la'
           } 
         }
      }
  }

# GitLab CI -Checkout git

stages: 
  - checkout

git-checkout:
  stage: checkout
  script:
    - git clone https://github.com/Pathnex/sample-java-app.git
    - cd sample-java-app
    - ls -la
