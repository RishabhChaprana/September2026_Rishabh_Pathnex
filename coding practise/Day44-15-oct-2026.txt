# Ansible - Create Multiple Users in Loops

---

- name: Create Multiple Pathnex users
  hosts: all
  become: yes

  vars:
    users:
      - dev1
      - dev2
      - dev3

  tasks:
    - name: Create users
      user:
        name: "{{ item }}"
        state: present
      loop: "{{ users }}"

# Kubernetes - Namespace + Deployment

apiVersion: v1
kind: Namespace
metadata:
  name: pathnex-namespace

---
apiVersion: app/v1
kind: Deployment
metadata:   
  name: pathnex-deploy
  namespace: pathnex-namespace
spec:
  replicas: 2
  selector:
    matchLables:
      app: pathnex
  template:
    metadata:
      lables:
        app: pathnex
    spec:
      containers:
        - name: app
          image: nginx

# Jenkins Pipline - Simulated Deloyment

pipeline {
    agent any
    parameters {
        choice(name: 'ENVIROMENT', choices: ['dev', 'staging', 'prod'], description: 'Deployment Enviroment')
    enviroment {
        INSTITUTE_NAME = "Pathnex"
    }
    stages {
        stage('Deploy') {
            steps {
                sh 'echo "Deployment $INSTITUTE_NAME application to $ENVIROMENT"'
            }
        }
    }
}
