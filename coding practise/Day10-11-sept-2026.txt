# Ansible - Create Docker Container for Multiple Application

# ```yaml

- name: Setup Docker Container for Multiple Application
  hosts: all
  become: yes
  tasks:
    - name: Pull Web App Image
      docker_image:
        name: my-web-app
        source: pull

    - name: Pull Database App Image
      docker_image:
        name: my-Database-app
        source: pull

    - name: Run web app container
      docker_container:
        name: web-app
        image: my-web-app
        stage: started
        published_port:
          - "3306:3306"

# Kubernets - Deploying StatefulSet

apiVersion: app/v1
kind: StatefulSet
metadata:
  name: pathnex-db
spec:
  serviceName: "pathnex-db"
  replicas: 3
  selector:
    matchLables:
      app: pathnex-db
  template:
    metadata:
      lables:
        app: pathnex-db
    spec:
      containers:
        - name: db
          image: mysql1:5.7
          volumeMounts:
            - mountPath: /var/lib/mysql
              name: mysql-storage
  volumeClaimTemplates:
    - metadata:
        name: mysql-stroage
      spec:
        accessModules: ["ReadWriteOnce"]
        resources:
          requests:
            stroage: 1Gi

# Jenkinsfile - Build and Deploy to Kubernetes

pipline {
   agent any
   enviroment {
       IMAGE_NAME = 'pathnex-web-app'
   }
   stages {
       stage('Build Docker Image') {
           steps {
               script {
                   docker.build("${env.IMAGE_NAME}")
               }
           }
        }
        stage('Push Docker Image') {
            steps {
                script {
                    docker.withRegistry('https://docker.io', 'docker-credentials') {
                    docker.image("${env.IMAGE_NAME}").push('latest')
                    }
                }
            }
        }
        stage('Deploy to Kubernetes') {
            steps {
                script {
                    sh 'kubectl apply -f deployment.yaml'
                }
            }
        }
    }
}
