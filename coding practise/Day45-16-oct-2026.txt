# Day45 - Basics

# Ansible - Install Nginx on Pathnex Server

- name: Install Nginx on Pathnex Server
  hosts: all
  become: yes

  tasks: 
    - name: Install nginx
      yum:
        name: nginx
        state: present
   
# Terraform Task - Create EC2 (c5.large)

provider "aws" {
  region = "us-east-1"
}

resource "aws_instance" "PathnexEC2" {
  ami		= "ami-0abcd1234abcd1234"
  instance_type = "c5.xlarge"
  
  tags = {
    Name = "Pathnex-EC2"
  }
}

# Kubernetes Tasks - Create Nginx Pod

apiVersion: v1
kind: Pod
metadata: 
  name: pathnex-pod
spec:
  containers:
    - name: web
      image: nginx:latest
      ports:
        - containerPort: 80
