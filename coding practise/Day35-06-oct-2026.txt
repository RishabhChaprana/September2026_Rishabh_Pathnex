# Jenkins Pipline - Debugging

pipeline {
    agent any
    stages {
        stage('Checkout') {
            steps {
                git branch: 'main',
                    url: 'https://github.com/Pathnex/sample-java-app.git'
            }
        }
        stage('Build') {
            steps {
                sh 'mvn clean compile'
             }
        }
        stage('Test') {
            steps {
                sh 'mvn test'
            }
        }
    }
    post {
        success {
            echo 'Pipline completed successfully'
        }
        failure {
            echo 'Pipline failed Check the console logs'
        }
        always {
            echo 'Pipline execution finished.'
        }
    }
}
