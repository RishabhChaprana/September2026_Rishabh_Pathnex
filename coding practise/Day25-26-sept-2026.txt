# Terraform - Create CloudWatch Alarms

resouce "aws_cloudwatch_metric_alarm" "pathnex_cpu_alarm" {
  alarm_name		= "pathnex-cpu-utilization-high"
  comparison_operator   = "GreaterThanThreshold"
  evaluation_periods	= "1"
  metric_name		= "CPUutilization"
  namespace		= "AWS/EC2"
  period		= "60"
  statistic		= "Average"
  threshold		= "80"

  dimensions = {
    InstanceId = aws_instance.pathnex_ec2.id
   }

   alarm_actions = [aws_sns_topic.pathnex_alerts.arn]
 }
resouce "aws_sns_topic" "pathnex_alerts" {
  name = "pathnex-alerts"
}

# Jenkinsfile - Conatiner Monitoring and Alerts in CI Pipeline

pipeline {
    agent any
    stages {
        stage('Monitor') {
            steps {
                script {
                    echo 'Trigging CloudWatch alarms if CPU exceeds 80%'
                    sh 'aws cloudwatch put-metric-data --namespace "AWS/EC2" --metric-name "CPUUtilizayion" --value 90'
                }
            }
        }
        stage('Test') {
            steps {
                echo 'Running tests...'
            }
        }
    }
}
