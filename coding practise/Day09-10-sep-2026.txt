# Day 09 -  Docker Integration with Jenkins & Github

# Ansible - Setup Docker Container for Nginx

- name: Setup Docker Container for Nginx
  hosts: all
  become: yes
  tasks: 
    - name: Pull Nginx image 
      docker_image:
        name: nginx
        source: pull
    - name: Run Nginx container
      docker_container:
        name: nginx_container
        image: nginx
        state: started
        published_port:
          - "8080:80"

# Kubernetes - Horizontal Pod Autoscaling (HPA)

apiVersion: app/v1
kind: Deployment
metadata:
  name: pathnex-deployment
spec:
  replicas: 1
  selector:
    matchLabels:
      app: pathnex-app
  template:
    metadata:
      lables:
   	app: pathnex-app
    spec:
      container:
        - name: nginx
          image: nginx
          ports:
            - containerPort: 80


---
apiVersion: autoscaling/v2
kind: HorizontalPodAutoscaler
metadata:
  name: pathnex-hpa
spec:
  scaleTargetRef:
    apiVersion: app/v1
    kind: Deployment
    name: pathnex-deployment
  minReplicas: 1
  maxReplicas: 10
  targetCPUUtilizationPercentage: 50
