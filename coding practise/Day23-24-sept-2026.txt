# Ansible - Deploy Lambda Function

- name: Deploy AWS Lambda Function
  hosts: localhost
  tasks: 
    - name: Create Lambda function
      command: aws lambda create-function --function-name pathnex-lambda --runtime python3.8 --role

# Terraform - Serverless API Gatway with Lambda Intergration

resource "aws_api_gateway_rest_api" "pathnex_api" {
  name 		= "pathnex-api"
  description	= "Pathnex API"
 }

resource "aws_lambda_function" "pathnex_lambda" {
  function_name		= "pathnex-lambda"
  runtime		= "python3.8"
  role			= aws_iam_role.lambda_exec_role.arn 
  handler		= "lambda_function.lambda_handler"
  filename		= "lambda.zip"
 }

resource "aws_api_gateway_integration" "lambda_intergration" {
  rest_api_id = aws_api_gatway_rest_api.pathnex_api.id
  resource_id = aws_api_gatway_resource.pathnex_resouce.id
  http_method = aws_api_datway_method.pathnex_method.http_method
  intergration_http_method = "POST"
  type = "AWS_PROXY"
  uri  = aws_lambda_function.pathnex_lambda.invoke_arn
 }

# Jenkinsfile - Deploy Serveless Application with Lambda

pipeline {
    agent any
    stages {
        stage('Build') {
            steps {
                echo 'Building Lambda Function...'
                sh 'zip -r lambda.zip' lambda/'
            }
        }
        stage('Deploy to AWS Lambda') {
            steps {
                script {
 		    sh 'aws lambda update-function-code --function-name pathnex-lambda --zip-file fileb://lambda.zip'
                }
            }
        }
    }
}

# Multi-Stage Docker Build

# Stage 1 - Build Stage

FROM maven:3.9-eclipse-temurin-17 AS builder
WORKDIR /opt/pathnex/java-build
COPY . /opt/pathnex/java-build/
RUN mvn clean package

# Stage 2 - Produce Stage
FROM tomcat:9
WORKDIR /usr/local/tomcat/webapp
COPY --from=builder /opt/pathnex/java-build/target/*.war /usr/local/tomcat/webapps/
EXPOSE 8080
CMD ["catalina.sh", "run"]

# Real Paths

/opt/pathnex/java-build
/usr/local/tomcat/webapps
